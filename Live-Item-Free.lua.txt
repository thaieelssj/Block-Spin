-- [[ Script: Live Items - Rastrreador de Armas para BlockSpin ]]
local Players = game:GetService("Players")
local TeleportService = game:GetService("TeleportService")
local LocalPlayer = Players.LocalPlayer
local PlayerGui = LocalPlayer:WaitForChild("PlayerGui")

-- Evitar duplicados si ya se ejecutó antes
if PlayerGui:FindFirstChild("LiveItemsGui") then
    PlayerGui.LiveItemsGui:Destroy()
end

-- Crear la Interfaz Principal (GUI)
local ScreenGui = Instance.new("ScreenGui")
ScreenGui.Name = "LiveItemsGui"
ScreenGui.Parent = PlayerGui

local MainFrame = Instance.new("Frame")
MainFrame.Size = UDim2.new(0, 320, 0, 420)
MainFrame.Position = UDim2.new(0.65, 0, 0.1, 0)
MainFrame.BackgroundColor3 = Color3.fromRGB(25, 25, 25)
MainFrame.BorderSizePixel = 0
MainFrame.Active = true
MainFrame.Draggable = true
MainFrame.Parent = ScreenGui

-- Título del Menú
local Title = Instance.new("TextLabel")
Title.Size = UDim2.new(1, 0, 0, 45)
Title.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
Title.Text = "⚡ LIVE ITEMS - BlockSpin ⚡"
Title.TextColor3 = Color3.fromRGB(255, 215, 0)
Title.TextScaled = true
Title.Parent = MainFrame

-- Contenedor con scroll para la lista de jugadores
local ScrollingFrame = Instance.new("ScrollingFrame")
ScrollingFrame.Size = UDim2.new(1, -10, 1, -95)
ScrollingFrame.Position = UDim2.new(0, 5, 0, 50)
ScrollingFrame.BackgroundColor3 = Color3.fromRGB(35, 35, 35)
ScrollingFrame.BorderSizePixel = 0
ScrollingFrame.CanvasSize = UDim2.new(0, 0, 0, 0)
ScrollingFrame.Parent = MainFrame

local UIListLayout = Instance.new("UIListLayout")
UIListLayout.SortOrder = Enum.SortOrder.LayoutOrder
UIListLayout.Padding = UDim.new(0, 5)
UIListLayout.Parent = ScrollingFrame

-- Función para actualizar la lista de jugadores y detectar armas
local function RefreshPlayerList()
    -- Limpiar lista anterior
    for _, child in ipairs(ScrollingFrame:GetChildren()) do
        if child:IsA("Frame") then
            child:Destroy()
        end
    end

    local totalHeight = 0

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            local playerName = player.Name
            local equippedWeapon = "Ninguna / Oculta"
            local isLegendary = false

            -- Revisar qué arma lleva equipada en su personaje
            if player.Character then
                for _, item in ipairs(player.Character:GetDescendants()) do
                    if item:IsA("Tool") or string.find(string.lower(item.ClassName), "tool") then
                        equippedWeapon = item.Name
                        local nameLower = string.lower(item.Name)
                        if string.find(nameLower, "legendary") or string.find(nameLower, "omega") or string.find(nameLower, "mythic") then
                            isLegendary = true
                        end
                        break
                    end
                end
            end

            -- Crear tarjeta visual para cada jugador (destacada si es legendaria/omega)
            local PlayerCard = Instance.new("Frame")
            PlayerCard.Size = UDim2.new(1, -10, 0, 55)
            PlayerCard.BackgroundColor3 = isLegendary and Color3.fromRGB(80, 60, 20) or Color3.fromRGB(45, 45, 45)
            PlayerCard.BorderSizePixel = 0
            PlayerCard.Parent = ScrollingFrame

            -- Si es legendaria, le añadimos un borde resaltado dorado
            if isLegendary then
                local UIStroke = Instance.new("UIStroke")
                UIStroke.Color = Color3.fromRGB(255, 215, 0)
                UIStroke.Thickness = 2
                UIStroke.Parent = PlayerCard
            end

            -- Texto con Nombre y Arma
            local InfoLabel = Instance.new("TextLabel")
            InfoLabel.Size = UDim2.new(0.62, 0, 1, 0)
            InfoLabel.Position = UDim2.new(0, 5, 0, 0)
            InfoLabel.BackgroundTransparency = 1
            InfoLabel.Text = "👤 " .. playerName .. "\n⚔️ " .. equippedWeapon
            InfoLabel.TextColor3 = isLegendary and Color3.fromRGB(255, 223, 0) or Color3.fromRGB(255, 255, 255)
            InfoLabel.TextSize = 12
            InfoLabel.TextXAlignment = Enum.TextXAlignment.Left
            InfoLabel.Parent = PlayerCard

            -- Botón para unirse al servidor de este usuario
            local JoinButton = Instance.new("TextButton")
            JoinButton.Size = UDim2.new(0.34, 0, 0.7, 0)
            JoinButton.Position = UDim2.new(0.64, 0, 0.15, 0)
            JoinButton.BackgroundColor3 = isLegendary and Color3.fromRGB(255, 170, 0) or Color3.fromRGB(70, 130, 180)
            JoinButton.Text = isLegendary and "¡UNIRME!" or "Seguir/Unir"
            JoinButton.TextColor3 = Color3.fromRGB(255, 255, 255)
            JoinButton.TextScaled = true
            JoinButton.Parent = PlayerCard

            -- Acción al hacer clic en el botón de unirse
            JoinButton.MouseButton1Click:Connect(function()
                JoinButton.Text = "Conectando..."
                pcall(function()
                    TeleportService:TeleportToPlaceInstance(game.PlaceId, game.JobId, LocalPlayer) 
                end)
            end)

            totalHeight = totalHeight + 60
        end
    end

    ScrollingFrame.CanvasSize = UDim2.new(0, 0, 0, totalHeight)
end

-- Botón inferior para actualizar la lista manualmente
local RefreshButton = Instance.new("TextButton")
RefreshButton.Size = UDim2.new(1, -10, 0, 35)
RefreshButton.Position = UDim2.new(0, 5, 1, -40)
RefreshButton.BackgroundColor3 = Color3.fromRGB(46, 139, 87)
RefreshButton.TextColor3 = Color3.fromRGB(255, 255, 255)
RefreshButton.Text = "🔄 Actualizar Live Items"
RefreshButton.TextScaled = true
RefreshButton.Parent = MainFrame

RefreshButton.MouseButton1Click:Connect(RefreshPlayerList)

-- Carga inicial del menú
RefreshPlayerList()
print("⚡ Live Items cargado correctamente en BlockSpin ⚡")
